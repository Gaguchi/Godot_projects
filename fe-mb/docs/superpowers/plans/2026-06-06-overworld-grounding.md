Steps use checkbox (`- [ ]`) syntax for tracking. Ralph-loop driver prompt: `docs/superpowers/plans/2026-06-06-overworld-grounding-loop-prompt.md`.

**Goal:** Make the FE_MB overworld map stop "floating in space" — ground every node in a coherent lit plane with contact shadows, carved roads, a caravan that walks the road, and (gated) a painterly per-region backdrop generated by ComfyUI.

**Architecture:** One coherent plane. Phase 1 is *zero-art* grounding on the existing backdrop (contact shadows, road casing, spline-walking token, asymmetric hearth vignette, state beacons) — SVG/CSS only, ships immediately. Phase 2 builds a depth-map rasterizer + a ComfyUI `backdrop` generator and bakes ONE proof backdrop that resolves a fallback ladder (Diorama → Lit Tabletop → Polished Parchment). Phase 3 rolls out the backdrop, node-derived value mask, fog-of-war, and the other two regions. Terrain is *terrain-only*; SVG owns every road/socket/beacon so node edits never force an art re-bake.

**Tech Stack:** TypeScript, React 19 (DOM + Tailwind), SVG, HTML Canvas, Zustand, Vitest, Playwright. Assets: local ComfyUI (SDXL `waiIllustrious_v160`, fp32/lowvram on a 6 GB GTX 1660), driven by `scripts/cli/comfy-gen.ts`.

```ts
// src/systems/overworldLayout.ts
export interface XY { x: number; y: number }
export interface PositionedNode { id: string; position: XY }

/** Linearly stretch node x-coords to [xMin,xMax], preserving relative spacing,
 *  order, and y. Pure; never mutates input. Topology (edges) is unaffected
 *  because edges reference ids, not positions. */
export function respreadNodes<T extends PositionedNode>(
  nodes: readonly T[],
  span: { xMin: number; xMax: number },
): T[] {
  const xs = nodes.map((n) => n.position.x)
  const lo = Math.min(...xs)
  const hi = Math.max(...xs)
  const range = hi - lo || 1
  return nodes.map((n) => ({
    ...n,
    position: { x: span.xMin + ((n.position.x - lo) / range) * (span.xMax - span.xMin), y: n.position.y },
  }))
}

/** Sample a quadratic-Bézier path string ("M x y Q cx cy x2 y2") into a polyline.
 *  Used to walk the token along the *actual* road, sampled once per hop. */
export function samplePolyline(path: string, segments = 24): XY[] {
  const m = path.match(/M\s*([\d.-]+)\s+([\d.-]+)\s+Q\s*([\d.-]+)\s+([\d.-]+)\s+([\d.-]+)\s+([\d.-]+)/)
  if (!m) return []
  const [x0, y0, cx, cy, x1, y1] = m.slice(1).map(Number) as [number, number, number, number, number, number]
  const pts: XY[] = []
  for (let i = 0; i <= segments; i++) {
    const t = i / segments
    const mt = 1 - t
    pts.push({ x: mt * mt * x0 + 2 * mt * t * cx + t * t * x1, y: mt * mt * y0 + 2 * mt * t * cy + t * t * y1 })
  }
  return pts
}

/** Point at fraction f∈[0,1] of a polyline's arc length (for even-speed travel). */
export function pointAtArcLength(pts: readonly XY[], f: number): { p: XY; angle: number } {
  if (pts.length === 0) return { p: { x: 0, y: 0 }, angle: 0 }
  if (pts.length === 1) return { p: pts[0]!, angle: 0 }
  const segLens: number[] = []
  let total = 0
  for (let i = 1; i < pts.length; i++) {
    const d = Math.hypot(pts[i]!.x - pts[i - 1]!.x, pts[i]!.y - pts[i - 1]!.y)
    segLens.push(d); total += d
  }
  let target = Math.max(0, Math.min(1, f)) * total
  for (let i = 0; i < segLens.length; i++) {
    if (target <= segLens[i]! || i === segLens.length - 1) {
      const t = segLens[i] ? target / segLens[i]! : 0
      const a = pts[i]!, b = pts[i + 1]!
      return { p: { x: a.x + (b.x - a.x) * t, y: a.y + (b.y - a.y) * t }, angle: Math.atan2(b.y - a.y, b.x - a.x) }
    }
    target -= segLens[i]!
  }
  return { p: pts[pts.length - 1]!, angle: 0 }
}
```

```tsx
import React from 'react'
import { ROAD, edgePath, hashEdgeId, edgeKey, type RoadState } from './roadStyle'

export function OverworldRoads({ edges, nodes, currentNodeId, plannedEdgeKeys, hoverEdgeKeys, activeEdgeKey, width, height }: {
  edges: { from: string; to: string }[]
  nodes: { id: string; position: { x: number; y: number } }[]
  currentNodeId: string
  plannedEdgeKeys: Set<string>
  hoverEdgeKeys: Set<string>
  activeEdgeKey: string | null
  width: number; height: number
}): React.JSX.Element {
  const pos = (id: string) => nodes.find((n) => n.id === id)?.position
  const paths = edges.map((e) => {
    const a = pos(e.from), b = pos(e.to)
    if (!a || !b) return null
    const key = edgeKey(e.from, e.to)
    const adjacent = e.from === currentNodeId || e.to === currentNodeId
    const state: RoadState = activeEdgeKey === key ? 'active'
      : plannedEdgeKeys.has(key) ? 'planned'
      : hoverEdgeKeys.has(key) ? 'hover'
      : adjacent ? 'adjacent' : 'distant'
    return { key, d: edgePath(a, b, hashEdgeId(e.from, e.to)), r: ROAD[state], state }
  }).filter(Boolean) as { key: string; d: string; r: typeof ROAD[RoadState]; state: RoadState }[]

  return (
    <svg width={width} height={height} className="absolute inset-0 pointer-events-none" style={{ overflow: 'visible' }}>
      <defs>
        <filter id="roadWorn" x="-20%" y="-20%" width="140%" height="140%">
          <feTurbulence type="fractalNoise" baseFrequency="0.012" numOctaves="2" seed="7" result="n" />
          <feDisplacementMap in="SourceGraphic" in2="n" scale="3" />
        </filter>
      </defs>
      {/* STATIC CASING — filtered once, no animation */}
      <g filter="url(#roadWorn)">
        {paths.map(({ key, d, r }) => (
          <g key={`c-${key}`}>
            <path d={d} fill="none" stroke="rgba(0,0,0,0.35)" strokeWidth={r.width + 5} strokeLinecap="round" transform="translate(0,1.5)" />
            <path d={d} fill="none" stroke="#caa766" strokeWidth={r.width + 3} strokeOpacity={0.5} strokeLinecap="round" transform="translate(0,-1)" />
            <path d={d} fill="none" stroke="#5b4126" strokeWidth={r.width} strokeOpacity={r.opacity ?? 1} strokeLinecap="round" />
            <path d={d} fill="none" stroke="#7a5836" strokeWidth={Math.max(0.6, r.width * 0.4)} strokeOpacity={0.6} strokeLinecap="round" />
          </g>
        ))}
      </g>
      {/* DYNAMIC STATE — unfiltered, animated */}
      <g>
        {paths.filter((p) => p.state === 'active' || p.state === 'planned' || p.state === 'hover').map(({ key, d, r }) => (
          <path key={`s-${key}`} d={d} fill="none" stroke={r.body} strokeWidth={Math.max(1.4, r.width - 1.4)}
            strokeLinecap="round"
            strokeDasharray={r.comet ? '14 9999' : r.dash ? '2 7' : undefined}
            className={r.comet ? 'road-comet' : undefined} />
        ))}
      </g>
    </svg>
  )
}
```

```tsx
import React, { useEffect, useRef, useState } from 'react'
import { samplePolyline, pointAtArcLength, type XY } from '../../../systems/overworldLayout'

export function Caravan({ hopPath, fromXY, durationMs, onArrive }: {
  hopPath: string | null     // edgePath() string for the active hop, or null when idle
  fromXY: XY                 // current node center (idle position)
  durationMs: number
  onArrive?: () => void
}): React.JSX.Element {
  const [f, setF] = useState(0)
  const polyRef = useRef<XY[]>([])
  useEffect(() => {
    if (!hopPath) { setF(0); return }
    polyRef.current = samplePolyline(hopPath, 28)   // sample ONCE per hop
    let raf = 0; let start = 0
    const tick = (t: number) => {
      if (!start) start = t
      const p = Math.min(1, (t - start) / durationMs)
      setF(p)
      if (p < 1) raf = requestAnimationFrame(tick)
      else onArrive?.()
    }
    raf = requestAnimationFrame(tick)
    return () => cancelAnimationFrame(raf)
  }, [hopPath, durationMs, onArrive])

  const here = hopPath && polyRef.current.length ? pointAtArcLength(polyRef.current, f) : { p: fromXY, angle: 0 }
  const bob = Math.sin(f * Math.PI * 8) * 1.5
  const lead = (off: number) => hopPath && polyRef.current.length ? pointAtArcLength(polyRef.current, Math.max(0, f - off)).p : fromXY

  return (
    <>
      {[0.12, 0.22].map((off, i) => {
        const p = lead(off)
        return <span key={i} aria-hidden className="absolute rounded-full"
          style={{ left: p.x - 5, top: p.y - 5, width: 10, height: 10, background: 'rgba(20,15,10,0.7)', filter: 'blur(0.5px)' }} />
      })}
      <div id="player-marker" className="absolute pointer-events-none z-20" style={{ left: here.p.x - 14, top: here.p.y - 18 + bob, width: 28, height: 30 }}>
        <span aria-hidden className="absolute left-1/2 rounded-[50%]" style={{ bottom: -3, width: 24, height: 8, transform: 'translateX(-50%)', background: 'radial-gradient(50% 50% at 50% 50%, rgba(0,0,0,0.55), transparent 75%)', filter: 'blur(1px)' }} />
        <div className="w-7 h-7 rounded-full bg-red-700 border-2 border-amber-50/90" style={{ boxShadow: '0 0 10px rgba(140,40,20,0.8)' }} />
      </div>
    </>
  )
}
```
